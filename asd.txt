let interactive = SigninLogs
    | where TimeGenerated > ago(30d)
    | where ResultType == 0
    | extend 
        deviceId  = tostring(DeviceDetail.deviceId),
        isManaged = tostring(DeviceDetail.isManaged),
        deviceOS  = tostring(DeviceDetail.operatingSystem)
    | project TimeGenerated, UserPrincipalName, AppDisplayName,
              IPAddress, deviceId, isManaged, deviceOS;
//
let nonInteractive = AADNonInteractiveUserSignInLogs
    | where TimeGenerated > ago(30d)
    | where ResultType == 0
    | extend parsedDevice = todynamic(tostring(DeviceDetail))
    | extend 
        deviceId  = tostring(parsedDevice.deviceId),
        isManaged = tostring(parsedDevice.isManaged),
        deviceOS  = tostring(parsedDevice.operatingSystem)
    | project TimeGenerated, UserPrincipalName, AppDisplayName,
              IPAddress, deviceId, isManaged, deviceOS;
//
union interactive, nonInteractive
| extend IsUnmanaged = isempty(deviceId) or deviceId == "" or isManaged != "true"
| summarize 
    TotalSignIns     = count(),
    UnmanagedSignIns = countif(IsUnmanaged),
    ManagedSignIns   = countif(not(IsUnmanaged)),
    FirstSeen        = min(TimeGenerated),
    LastSeen         = max(TimeGenerated),
    OSTypes          = make_set(deviceOS, 10),
    IPs              = make_set(iff(IsUnmanaged, IPAddress, ""), 25),
    Apps             = make_set(iff(IsUnmanaged, AppDisplayName, ""), 25)
  by UserPrincipalName
| where UnmanagedSignIns > 0
| extend OSTypes = set_difference(OSTypes, dynamic([""]))
| extend IPs = set_difference(IPs, dynamic([""]))
| extend Apps = set_difference(Apps, dynamic([""]))
| extend UnmanagedPct = round(100.0 * UnmanagedSignIns / TotalSignIns, 1)
| project UserPrincipalName, UnmanagedPct, ManagedSignIns, UnmanagedSignIns,
          FirstSeen, LastSeen, OSTypes, IPs, Apps
| order by UnmanagedPct desc, UnmanagedSignIns desc
