let interactive = SigninLogs
    | where TimeGenerated > ago(30d)
    | where ResultType == 0
    | extend 
        deviceId    = tostring(DeviceDetail.deviceId),
        isManaged   = tostring(DeviceDetail.isManaged),
        isCompliant = tostring(DeviceDetail.isCompliant),
        deviceOS    = tostring(DeviceDetail.operatingSystem),
        SignInType   = "Interactive",
        origRequestId = tostring(OriginalRequestId)
    | project TimeGenerated, UserPrincipalName, AppDisplayName, 
              IPAddress, deviceId, isManaged, isCompliant, 
              deviceOS, SignInType, Id, origRequestId;
//
let nonInteractive = AADNonInteractiveUserSignInLogs
    | where TimeGenerated > ago(30d)
    | where ResultType == 0
    | extend parsedDevice = todynamic(tostring(DeviceDetail))
    | extend 
        deviceId    = tostring(parsedDevice.deviceId),
        isManaged   = tostring(parsedDevice.isManaged),
        isCompliant = tostring(parsedDevice.isCompliant),
        deviceOS    = tostring(parsedDevice.operatingSystem),
        SignInType   = "NonInteractive",
        origRequestId = tostring(OriginalRequestId)
    | project TimeGenerated, UserPrincipalName, AppDisplayName, 
              IPAddress, deviceId, isManaged, isCompliant, 
              deviceOS, SignInType, Id, origRequestId;
//
let deviceLookup = union interactive, nonInteractive
    | where isnotempty(deviceId) and deviceId != ""
    | project Id, UserPrincipalName, deviceId, 
              isManaged, isCompliant, IPAddress, TimeGenerated;
//
let missingDevice = union interactive, nonInteractive
    | where isempty(deviceId) or deviceId == "";
//
// Layer 1: Resolve via OriginalRequestId -> Id of the original sign-in
let resolvedLayer1 = missingDevice
    | where isnotempty(origRequestId)
    | join kind=leftouter (
        deviceLookup 
        | project-rename origRequestId = Id, 
                         resolvedDeviceId = deviceId,
                         resolvedManaged = isManaged
    ) on origRequestId, UserPrincipalName
    | extend Resolved = isnotempty(resolvedDeviceId);
//
// Layer 2: IP + time window for still unresolved
let unresolvedAfterL1 = resolvedLayer1 
    | where not(Resolved)
    | project-away resolvedDeviceId, resolvedManaged, Resolved;
let resolvedLayer2 = unresolvedAfterL1
    | join kind=leftouter (
        deviceLookup
        | project UserPrincipalName, IPAddress, 
                  resolvedDeviceId = deviceId, 
                  resolvedManaged = isManaged, 
                  ManagedTime = TimeGenerated
    ) on UserPrincipalName, IPAddress
    | where abs(datetime_diff('minute', TimeGenerated, ManagedTime)) < 30
    | extend Resolved = isnotempty(resolvedDeviceId)
    | summarize arg_max(Resolved, *) by TimeGenerated, UserPrincipalName, Id;
//
let trulyUnresolved = resolvedLayer2
    | where not(Resolved);
//
trulyUnresolved
| summarize 
    FirstSeen       = min(TimeGenerated),
    LastSeen        = max(TimeGenerated),
    OSTypes         = make_set(deviceOS),
    IPs             = make_set(IPAddress, 25),
    SignInCount     = count(),
    Apps            = make_set(AppDisplayName)
  by UserPrincipalName
| extend OSTypes = set_difference(OSTypes, dynamic([""]))
| order by SignInCount desc
